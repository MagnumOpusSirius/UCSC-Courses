//if ease if 

function checkGrade(score){
    if(score>=90){
        console.log("excellent");
    }else if (score>=70){
        console.log("good"); 
    }else if (score>=50){
        console.log('pass')
    }else {
        console.log('Fail')
    }

}

checkGrade(85);
checkGrade(95);
checkGrade(55);
checkGrade(40);
